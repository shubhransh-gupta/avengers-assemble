🚀 30-SECOND QUICKSTART

Assemble in Under 30 Seconds

Zero config required. Just npx and go.

01 // RUN A MISSION

# Zero install — just run with npx
$ npx avengers-assemble assemble \
  "Build JWT auth + unit tests + Docker"

# Or install globally
$ npm install -g avengers-assemble
$ avengers assemble "Your mission"

02 // OPEN THE HUD

# Launch Stark Tower Mission Control
$ avengers hud

# Opens http://localhost:3000
# Live telemetry, hero cards, comms

# Or check team readiness
$ avengers status

03 // CONFIGURE PROVIDERS

# Copy the config template
$ cp stark.config.example.json stark.config.json

"providers": {
  "claude-code": { "apiKey": "sk-ant-..." },
  "gemini": { "apiKey": "AIza..." },
  "grok": { "apiKey": "xai-..." }
}

04 // LOCAL DEV SETUP

$ git clone gh:shubhransh-gupta/avengers-assemble
$ cd avengers-assemble
$ npm install
$ npm test # 6/6 ✔
$ npm start # Opens HUD
